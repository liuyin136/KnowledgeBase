   111→function timeAgo(iso: string): string {
   112→  const then = new Date(iso).getTime();
   113→  const now = Date.now();
   114→  const diffSec = Math.max(1, Math.floor((now - then) / 1000));
   115→  if (diffSec < 60) return `${diffSec}s ago`;
   116→  const diffMin = Math.floor(diffSec / 60);
   117→  if (diffMin < 60) return `${diffMin}m ago`;
   118→  const diffHr = Math.floor(diffMin / 60);
   119→  if (diffHr < 24) return `${diffHr}h ago`;
   120→  const diffDay = Math.floor(diffHr / 24);
   121→  if (diffDay < 7) return `${diffDay}d ago`;
   122→  const diffWk = Math.floor(diffDay / 7);
   123→  if (diffWk < 5) return `${diffWk}w ago`;
   124→  return new Date(iso).toLocaleDateString();
   125→}
   126→
   127→function formatMs(ms: number): string {
   128→  if (!ms || ms <= 0) return "—";
   129→  if (ms < 1000) return `${Math.round(ms)}ms`;
   130→  return `${(ms / 1000).toFixed(2)}s`;
   131→}
   132→
   133→function statusTone(status: string): { className: string; label: string } {
   134→  switch (status) {
   135→    case "completed":
   136→      return { label: "completed", className: "border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400" };
   137→    case "failed":
   138→      return { label: "failed", className: "border-red-500/30 bg-red-500/10 text-red-700 dark:text-red-400" };
   139→    case "running":
   140→      return { label: "running", className: "border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-400" };
   141→    case "pending":
   142→    default:
   143→      return { label: status || "pending", className: "border-slate-500/30 bg-slate-500/10 text-slate-700 dark:text-slate-400" };
   144→  }
   145→}
   146→
   147→function truncate(text: string, n: number): string {
   148→  if (!text) return "";
   149→  return text.length > n ? text.slice(0, n) + "…" : text;
   150→}
