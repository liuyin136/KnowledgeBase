   110→// ─── Helpers ─────────────────────────────────────────────────────────────────
   111→function relativeTime(d: string | Date | null | undefined): string {
   112→  if (!d) return "—";
   113→  try {
   114→    return formatDistanceToNow(new Date(d), { addSuffix: true });
   115→  } catch {
   116→    return "—";
   117→  }
   118→}
   119→
   120→function fmtMs(ms: number | null | undefined): string {
   121→  if (ms === null || ms === undefined || Number.isNaN(ms)) return "—";
   122→  if (ms >= 1000) return `${(ms / 1000).toFixed(2)}s`;
   123→  return `${Math.round(ms)}ms`;
   124→}
   125→
   126→function fmtNum(n: number | null | undefined, digits = 0): string {
   127→  if (n === null || n === undefined || Number.isNaN(n)) return "—";
   128→  return n.toLocaleString(undefined, { maximumFractionDigits: digits });
   129→}

   135→function statusBadge(status: ExperimentStatus) {
   136→  const map: Record<ExperimentStatus, { cls: string; icon: React.ReactNode; label: string }> = {
   137→    completed: {
   138→      cls: "border-transparent bg-primary/15 text-primary",
   139→      icon: <CheckCircle2 className="h-3 w-3" />,
   140→      label: "completed",
   141→    },
   142→    failed: {
   143→      cls: "border-transparent bg-destructive/15 text-destructive",
   144→      icon: <XCircle className="h-3 w-3" />,
   145→      label: "failed",
   146→    },
   147→    running: {
   148→      cls: "border-transparent bg-amber-500/15 text-amber-600 dark:text-amber-400",
   149→      icon: <Loader2 className="h-3 w-3 animate-spin" />,
   150→      label: "running",
   151→    },
   152→    pending: {
   153→      cls: "border-transparent bg-slate-500/15 text-slate-600 dark:text-slate-300",
   154→      icon: <CircleDashed className="h-3 w-3" />,
   155→      label: "pending",
   156→    },
   157→  };
   158→  const s = map[status] ?? map.pending;
   159→  return (
   160→    <Badge variant="outline" className={cn("gap-1 text-[10px] font-medium", s.cls)}>
   161→      {s.icon}
   162→      {s.label}
   163→    </Badge>
   164→  );
   165→}
   166→
   167→function approachBadge(approach: string) {
   168→  if (approach === "Query") {
   169→    return <Badge variant="secondary" className="text-[10px]">Query</Badge>;
   170→  }
   171→  if (approach === "LongText") {
   172→    return <Badge variant="default" className="text-[10px]">LongText</Badge>;
   173→  }
   174→  if (approach === "ChildChunk") {
   175→    return <Badge className="text-[10px] bg-primary/85">ChildChunk</Badge>;
   176→  }
   177→  return <Badge variant="outline" className="text-[10px]">{approach}</Badge>;
   178→}

   180→function chunkMethodBadge(method: string) {
   181→  if (method === "N/A" || !method) return <span className="text-xs text-muted-foreground">N/A</span>;
   182→  return <Badge variant="outline" className="text-[10px]">{method}</Badge>;

   218→    <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-4 gap-2.5">
   219→      <StatCard icon={Boxes} label="Total chunks" value={fmtNum(exp.totalChunks)} mono />
   220→      <StatCard
   221→        icon={Hash}
   222→        label="Avg tokens / chunk"
   223→        value={fmtNum(exp.avgTokensPerChunk, 1)}
   224→        mono
   225→      />
   226→      <StatCard icon={Timer} label="Total time" value={fmtMs(exp.totalTimeMs)} mono />
   227→      <StatCard icon={Gauge} label="Status" value={statusBadge(exp.status)} />
   228→      <StatCard icon={Layers} label="Embedding" value={approachBadge(exp.embeddingApproach)} />
   229→      <StatCard icon={Layers} label="Chunk method" value={chunkMethodBadge(exp.chunkMethod)} />

   232→        icon={FileText}

   236→            {exp.sourceFile ?? "—"}
   237→          </span>
   238→        }
   239→      />
   240→      {chunks && (
   241→        <>
   242→          <StatCard icon={Timer} label="Σ Chunking ms" value={fmtMs(totalChunkingMs)} mono />

   365→// ─── Chunk Browser (detail mode) ────────────────────────────────────────────
   366→function ChunkBrowser({ experimentId }: { experimentId: string }) {
   367→  const { data, isLoading, isError, refetch } = useQuery({
   368→    queryKey: ["experiment", experimentId, "chunks"],
   369→    queryFn: () => api.experiments.chunks(experimentId),
   370→  });
   371→  const [active, setActive] = React.useState<ChunkRow | null>(null);
   372→  const [open, setOpen] = React.useState(false);
   373→
   374→  const chunks = ((data?.items ?? []) as ChunkRow[]);
   375→
   376→  const openRow = (c: ChunkRow) => {
   377→    setActive(c);
   378→    setOpen(true);
   379→  };
   380→
   381→  return (
   382→    <Card>
   383→      <CardHeader>
   384→        <CardTitle className="flex items-center gap-2 text-base">
   385→          <Boxes className="h-4 w-4 text-primary" />
   386→          Chunk Browser
   387→        </CardTitle>
   388→        <CardDescription className="text-xs">
   389→          {chunks.length} chunk{chunks.length === 1 ? "" : "s"} · click a row to inspect
   390→        </CardDescription>
   391→        <CardAction>
   392→          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => refetch()} aria-label="Refresh chunks">
   393→            <RefreshCw className="h-3.5 w-3.5" />
   394→          </Button>
   395→        </CardAction>
   396→      </CardHeader>
   397→      <CardContent>
   398→        {isLoading ? (
   399→          <div className="space-y-2">
   400→            {[0, 1, 2, 3].map((i) => (
   401→              <Skeleton key={i} className="h-9 w-full" />
   402→            ))}
   403→          </div>
   404→        ) : isError ? (
   405→          <div className="flex items-center gap-2 text-xs text-destructive">
   406→            <AlertCircle className="h-4 w-4" /> Failed to load chunks.
   407→          </div>
   408→        ) : chunks.length === 0 ? (
   409→          <div className="rounded-md border border-dashed p-6 text-center text-xs text-muted-foreground">
   410→            No chunks recorded for this experiment.
   411→          </div>
   412→        ) : (
   413→          <div className="rounded-md border max-h-96 overflow-y-auto thin-scroll">
   414→            <Table>
   415→              <TableHeader className="sticky top-0 bg-background z-10">
   416→                <TableRow>
   417→                  <TableHead className="w-[50px]">#</TableHead>
   418→                  <TableHead className="w-[110px]">Method</TableHead>
   419→                  <TableHead className="w-[110px]">Embedding</TableHead>
   420→                  <TableHead className="w-[70px] text-right">Tokens</TableHead>
   421→                  <TableHead className="w-[80px] text-right">Chunk ms</TableHead>
   422→                  <TableHead className="w-[80px] text-right">Embed ms</TableHead>
   423→                  <TableHead className="w-[110px]">Section</TableHead>
   424→                  <TableHead className="min-w-[260px]">Preview</TableHead>
   425→                  <TableHead className="w-[40px]"></TableHead>
   426→                </TableRow>
   427→              </TableHeader>
   428→              <TableBody>
   429→                {chunks.map((c) => (
   430→                  <TableRow
   431→                    key={c.chunkId}
   432→                    className="group cursor-pointer"
   433→                    onClick={() => openRow(c)}
   434→                  >
   435→                    <TableCell className="font-mono text-xs">{c.chunkIndex}</TableCell>
   436→                    <TableCell>{chunkMethodBadge(c.chunkMethod)}</TableCell>
   437→                    <TableCell>
   438→                      <Badge variant="outline" className="text-[10px]">{c.embeddingMethod}</Badge>
   439→                    </TableCell>
   440→                    <TableCell className="text-right font-mono text-xs">{fmtNum(c.tokenCount)}</TableCell>
   441→                    <TableCell className="text-right font-mono text-xs">{fmtMs(c.chunkingTimeMs)}</TableCell>
   442→                    <TableCell className="text-right font-mono text-xs">{fmtMs(c.embeddingTimeMs)}</TableCell>
   443→                    <TableCell className="text-[10px] text-muted-foreground truncate max-w-[110px]" title={c.section ?? ""}>
   444→                      {c.section ?? "—"}
   445→                    </TableCell>
   446→                    <TableCell className="text-xs text-muted-foreground line-clamp-1 max-w-[280px]">
   447→                      {c.textPreview}
   448→                    </TableCell>
   449→                    <TableCell>
   450→                      <Eye className="h-3 w-3 text-muted-foreground opacity-0 group-hover:opacity-100 transition-opacity" />
   451→                    </TableCell>
   452→                  </TableRow>
   453→                ))}
   454→              </TableBody>
   455→            </Table>
   456→          </div>
   457→        )}
   458→      </CardContent>
   459→      <ChunkInspectorSheet chunk={active} open={open} onOpenChange={setOpen} />
   460→    </Card>
   461→  );
   462→}

   464→// ─── Detail Mode ────────────────────────────────────────────────────────────
   465→function DetailMode({
   466→  experimentId,
   467→  onBack,
   468→  onCompareWith,
   469→}: {
   470→  experimentId: string;
   471→  onBack: () => void;
   472→  onCompareWith: (otherId: string) => void;
   473→}) {
   474→  const { data: exp, isLoading, isError, refetch } = useQuery({
   475→    queryKey: ["experiment", experimentId],
   476→    queryFn: () => api.experiments.get(experimentId),
   477→  });
   478→  const [pickerOpen, setPickerOpen] = React.useState(false);
   479→  const [otherId, setOtherId] = React.useState<string>("");
   480→
   481→  const { data: listData } = useQuery({
   482→    queryKey: ["experiments", "list", { kind: "all", page: 1, pageSize: 100 }],
   483→    queryFn: () => api.experiments.list({ page: 1, pageSize: 100 }),
   484→    enabled: pickerOpen,
   485→  });
   486→  const candidates = (listData?.items ?? []).filter(
   487→    (e: Experiment) => e.id !== experimentId
   488→  );
   489→
   490→  if (isLoading) {
   491→    return (
   492→      <div className="space-y-4">
   493→        <Skeleton className="h-8 w-32" />
   494→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-2.5">
   495→          {[0, 1, 2, 3, 4, 5, 6, 7].map((i) => (
   496→            <Skeleton key={i} className="h-16 w-full" />
   497→          ))}
   498→        </div>
   499→        <Skeleton className="h-64 w-full" />
   500→      </div>
   501→    );
   502→  }
   503→
   504→  if (isError || !exp) {
   505→    return (
   506→      <Card>
   507→        <CardContent className="pt-6">
   508→          <div className="flex items-center gap-2 text-sm text-destructive">
   509→            <AlertCircle className="h-4 w-4" /> Failed to load experiment.
   510→          </div>
   511→          <div className="flex gap-2 mt-3">
   512→            <Button variant="outline" size="sm" onClick={onBack}>← Back</Button>
   513→            <Button variant="outline" size="sm" onClick={() => refetch()}>Retry</Button>
   514→          </div>
   515→        </CardContent>
   516→      </Card>
   517→    );
   518→  }
   519→
   520→  return (
   521→    <div className="space-y-4">
   522→      <div className="flex items-center justify-between gap-2 flex-wrap">
   523→        <Button variant="ghost" size="sm" onClick={onBack} className="gap-1">
   524→          <ArrowLeft className="h-4 w-4" /> Back to list
   525→        </Button>
   526→        <Button variant="outline" size="sm" onClick={() => setPickerOpen(true)} className="gap-1.5">
   527→          <GitCompareArrows className="h-3.5 w-3.5" /> Compare with…
   528→        </Button>
   529→      </div>
   530→
   531→      <Card>
   532→        <CardHeader>
   533→          <CardTitle className="flex items-center gap-2 text-base">
   534→            <FlaskConical className="h-4 w-4 text-primary" />
   535→            <span className="truncate">{exp.description}</span>
   536→          </CardTitle>
   537→          <CardDescription className="text-xs flex flex-wrap items-center gap-x-2 gap-y-1">
   538→            <span className="font-mono">{exp.id}</span>
   539→            <span aria-hidden>·</span>
   540→            <span className="flex items-center gap-1">

   752→// ─── Comparison Table ───────────────────────────────────────────────────────
   753→function ComparisonTable({
   754→  a,
   755→  b,
   756→  chunksA,
   757→  chunksB,
   758→}: {
   759→  a: Experiment;
   760→  b: Experiment;
   761→  chunksA: ChunkRow[];
   762→  chunksB: ChunkRow[];
   763→}) {
   764→  const totalChunkingA = chunksA.reduce((s, c) => s + (c.chunkingTimeMs || 0), 0);
   765→  const totalChunkingB = chunksB.reduce((s, c) => s + (c.chunkingTimeMs || 0), 0);
   766→  const totalEmbedA = chunksA.reduce((s, c) => s + (c.embeddingTimeMs || 0), 0);
   767→  const totalEmbedB = chunksB.reduce((s, c) => s + (c.embeddingTimeMs || 0), 0);
   768→
   769→  type Row = {
   770→    metric: string;
   771→    a: React.ReactNode;
   772→    b: React.ReactNode;
   773→    delta?: { value: string; positive: boolean } | null;
   774→  };
   775→
   776→  const numDelta = (av: number | null | undefined, bv: number | null | undefined, fmt: (n: number) => string) => {
   777→    if (av == null || bv == null) return null;
   778→    const d = bv - av;
   779→    const sign = d > 0 ? "+" : "";
   780→    return { value: `${sign}${fmt(d)}`, positive: d >= 0 };
   781→  };
   782→
   783→  const rows: Row[] = [
   784→    {
   785→      metric: "Embedding approach",
   786→      a: approachBadge(a.embeddingApproach),
   787→      b: approachBadge(b.embeddingApproach),
   788→    },
   789→    {
   790→      metric: "Chunk method",
   791→      a: chunkMethodBadge(a.chunkMethod),
   792→      b: chunkMethodBadge(b.chunkMethod),
   793→    },
   794→    {
   795→      metric: "Total chunks",
   796→      a: fmtNum(a.totalChunks),
   797→      b: fmtNum(b.totalChunks),
   798→      delta: numDelta(a.totalChunks, b.totalChunks, (n) => fmtNum(n)),
   799→    },
   800→    {
   801→      metric: "Avg tokens / chunk",
   802→      a: fmtNum(a.avgTokensPerChunk, 1),
   803→      b: fmtNum(b.avgTokensPerChunk, 1),
   804→      delta: numDelta(a.avgTokensPerChunk, b.avgTokensPerChunk, (n) => n.toFixed(1)),
   805→    },
   806→    {
   807→      metric: "Total time (ms)",
   808→      a: fmtMs(a.totalTimeMs),
   809→      b: fmtMs(b.totalTimeMs),
   810→      delta: numDelta(a.totalTimeMs, b.totalTimeMs, (n) => `${Math.round(n)}ms`),
   811→    },
   812→    {
   813→      metric: "Σ Chunking time (ms)",
   814→      a: chunksA.length ? fmtMs(totalChunkingA) : "—",
   815→      b: chunksB.length ? fmtMs(totalChunkingB) : "—",
   816→      delta: chunksA.length && chunksB.length ? numDelta(totalChunkingA, totalChunkingB, (n) => `${Math.round(n)}ms`) : null,
   817→    },
   818→    {
   819→      metric: "Σ Embedding time (ms)",
   820→      a: chunksA.length ? fmtMs(totalEmbedA) : "—",
   821→      b: chunksB.length ? fmtMs(totalEmbedB) : "—",
   822→      delta: chunksA.length && chunksB.length ? numDelta(totalEmbedA, totalEmbedB, (n) => `${Math.round(n)}ms`) : null,
   823→    },
   824→    {
   825→      metric: "Status",
   826→      a: statusBadge(a.status),
   827→      b: statusBadge(b.status),
   828→    },
   829→    {
   830→      metric: "Source file",
   831→      a: <span className="text-xs truncate block max-w-[180px]" title={a.sourceFile ?? ""}>{a.sourceFile ?? "—"}</span>,
   832→      b: <span className="text-xs truncate block max-w-[180px]" title={b.sourceFile ?? ""}>{b.sourceFile ?? "—"}</span>,
   833→    },
   834→  ];
   835→
   836→  return (
   837→    <Card>
   838→      <CardHeader>
   839→        <CardTitle className="text-base">Metric Comparison</CardTitle>
   840→        <CardDescription className="text-xs">
   841→          Δ = B − A. <span className="text-primary">teal</span> for positive, muted for negative.
   842→        </CardDescription>
   843→      </CardHeader>
   844→      <CardContent>
   845→        <div className="rounded-md border overflow-hidden">
   846→          <Table>
   847→            <TableHeader>
   848→              <TableRow>
   849→                <TableHead className="w-[180px]">Metric</TableHead>
   850→                <TableHead className="w-[20%]">Experiment A</TableHead>
   851→                <TableHead className="w-[20%]">Experiment B</TableHead>
   852→                <TableHead className="w-[80px] text-right">Δ</TableHead>
   853→              </TableRow>
   854→            </TableHeader>
   855→            <TableBody>
   856→              {rows.map((r) => (
   857→                <TableRow key={r.metric}>
   858→                  <TableCell className="text-xs font-medium text-muted-foreground">{r.metric}</TableCell>
   859→                  <TableCell>{r.a}</TableCell>
   860→                  <TableCell>{r.b}</TableCell>
   861→                  <TableCell className="text-right">
   862→                    {r.delta ? (
   863→                      <span
   864→                        className={cn(
   865→                          "font-mono text-xs",
   866→                          r.delta.positive ? "text-primary" : "text-muted-foreground"
   867→                        )}
   868→                      >
   869→                        {r.delta.value}
   870→                      </span>
   871→                    ) : (
   872→                      <span className="text-muted-foreground text-xs">—</span>
   873→                    )}
   874→                  </TableCell>
   875→                </TableRow>
   876→              ))}
   877→            </TableBody>
   878→          </Table>
   879→        </div>
   880→      </CardContent>
   881→    </Card>
   882→  );
   883→}

   885→// ─── Bar Comparison ─────────────────────────────────────────────────────────
   886→function BarComparison({
   887→  a,
   888→  b,
   889→  chunksA,
   890→  chunksB,
   891→}: {
   892→  a: Experiment;
   893→  b: Experiment;
   894→  chunksA: ChunkRow[];
   895→  chunksB: ChunkRow[];
   896→}) {
   897→  if (chunksA.length === 0 && chunksB.length === 0) return null;
   898→
   899→  const maxChunks = Math.max(chunksA.length, chunksB.length, 1);
   900→  const maxTokens = Math.max(a.avgTokensPerChunk, b.avgTokensPerChunk, 1);
   901→
   902→  const Bar = ({
   903→    label,
   904→    valueA,
   905→    valueB,
   906→    fmtVal,
   907→  }: {
   908→    label: string;
   909→    valueA: number;
   910→    valueB: number;
   911→    fmtVal: (n: number) => string;
   912→  }) => {
   913→    const max = Math.max(valueA, valueB, 1);
   914→    return (
   915→      <div className="space-y-1.5">
   916→        <div className="text-[10px] uppercase tracking-wide text-muted-foreground">{label}</div>
   917→        <div className="space-y-1">
   918→          <BarRow label="A" value={valueA} max={max} fmtVal={fmtVal} />
   919→          <BarRow label="B" value={valueB} max={max} fmtVal={fmtVal} />
   920→        </div>
   921→      </div>
   922→    );
   923→  };
   924→
   925→  const BarRow = ({
   926→    label,
   927→    value,
   928→    max,
   929→    fmtVal,
   930→  }: {
   931→    label: string;
   932→    value: number;
   933→    max: number;
   934→    fmtVal: (n: number) => string;
   935→  }) => (
   936→    <div className="flex items-center gap-2">
   937→      <span className="text-[10px] font-mono w-3 text-muted-foreground">{label}</span>
   938→      <div className="flex-1 h-3 rounded bg-muted overflow-hidden">
   939→        <div
   940→          className={cn(
   941→            "h-full rounded transition-all",
   942→            label === "A" ? "bg-primary/70" : "bg-primary"
   943→          )}
   944→          style={{ width: `${Math.max(2, (value / max) * 100)}%` }}
   945→        />
   946→      </div>
   947→      <span className="text-[10px] font-mono w-14 text-right">{fmtVal(value)}</span>
   948→    </div>
   949→  );
   950→
   951→  return (
   952→    <Card>
   953→      <CardHeader>
   954→        <CardTitle className="text-base">Distribution</CardTitle>
   955→        <CardDescription className="text-xs">
   956→          Quick visual comparison of chunk count and average tokens.
   957→        </CardDescription>
   958→      </CardHeader>
   959→      <CardContent className="space-y-4">
   960→        <Bar
   961→          label="Chunk count"
   962→          valueA={chunksA.length}
   963→          valueB={chunksB.length}
   964→          fmtVal={(n) => fmtNum(n)}
   965→        />
   966→        <Bar
   967→          label="Avg tokens / chunk"
   968→          valueA={a.avgTokensPerChunk}
   969→          valueB={b.avgTokensPerChunk}
   970→          fmtVal={(n) => fmtNum(n, 1)}
   971→        />
   972→        <div className="text-[10px] text-muted-foreground">
   973→          Max chunks (A: {chunksA.length}, B: {chunksB.length}) · Max avg tokens: {fmtNum(maxTokens, 1)}
   974→        </div>
   975→      </CardContent>
   976→    </Card>
   977→  );
   978→}

  1129→        {/* Pagination */}

  1153→          </div>

  1155→      </CardContent>
  1156→    </Card>
