    66→// ─── helpers ─────────────────────────────────────────────────────────────────
    67→function relativeTime(d: string | Date | null | undefined): string {
    68→  if (!d) return "—";
    69→  try {
    70→    return formatDistanceToNow(new Date(d), { addSuffix: true });
    71→  } catch {
    72→    return "—";
    73→  }
    74→}
    75→
    76→function fmtScore(n: number | null | undefined): string {
    77→  if (n === null || n === undefined || Number.isNaN(n)) return "—";
    78→  return n.toFixed(3);
    79→}
    80→
    81→function fmtMs(ms: number | null | undefined): string {
    82→  if (ms === null || ms === undefined || Number.isNaN(ms)) return "—";
    83→  if (ms >= 1000) return `${(ms / 1000).toFixed(2)}s`;
    84→  return `${Math.round(ms)}ms`;
    85→}

    90→// ─── Create Cart ─────────────────────────────────────────────────────────────
    91→function CreateCartCard({ onCreated }: { onCreated: (id: string) => void }) {
    92→  const qc = useQueryClient();
    93→  const [name, setName] = React.useState("");
    94→  const [description, setDescription] = React.useState("");
    95→
    96→  const createMut = useMutation({
    97→    mutationFn: () =>
    98→      api.memoryCarts.create({
    99→        name: name.trim(),
   100→        description: description.trim() || undefined,
   101→      }),
   102→    onSuccess: (data) => {
   103→      toast.success("Cart created", { description: `"${name.trim()}"` });
   104→      qc.invalidateQueries({ queryKey: ["carts"] });
   105→      setName("");
   106→      setDescription("");
   107→      onCreated(data.id);
   108→    },
   109→    onError: (e: unknown) => {
   110→      const msg = e instanceof Error ? e.message : "Failed to create cart";
   111→      toast.error("Create failed", { description: msg });
   112→    },
   113→  });
   114→
   115→  const canSubmit = name.trim().length > 0 && !createMut.isPending;
   116→
   117→  return (
   118→    <Card>
   119→      <CardHeader>
   120→        <CardTitle className="flex items-center gap-2 text-base">
   121→          <Plus className="h-4 w-4 text-primary" />
   122→          Create Cart
   123→        </CardTitle>
   124→        <CardDescription className="text-xs">
   125→          A cart groups curated memories for export or sharing.
   126→        </CardDescription>
   127→      </CardHeader>
   128→      <CardContent className="space-y-3">
   129→        <div className="space-y-1.5">
   130→          <Label htmlFor="cart-name" className="text-xs">Name</Label>
   131→          <Input
   132→            id="cart-name"
   133→            placeholder="e.g. Best hybrid-search hits"
   134→            value={name}
   135→            onChange={(e) => setName(e.target.value)}
   136→            onKeyDown={(e) => {
   137→              if (e.key === "Enter" && canSubmit) createMut.mutate();
   138→            }}
   139→            disabled={createMut.isPending}
   140→          />
   141→        </div>
   142→        <div className="space-y-1.5">
   143→          <Label htmlFor="cart-desc" className="text-xs">
   144→            Description <span className="text-muted-foreground">(optional)</span>
   145→          </Label>
   146→          <Textarea
   147→            id="cart-desc"
   148→            placeholder="What this cart is for…"
   149→            value={description}
   150→            onChange={(e) => setDescription(e.target.value)}
   151→            rows={2}
   152→            disabled={createMut.isPending}
   153→            className="resize-none"
   154→          />
   155→        </div>
   156→        <Button
   157→          className="w-full"
   158→          size="sm"
   159→          disabled={!canSubmit}
   160→          onClick={() => createMut.mutate()}
   161→        >
   162→          {createMut.isPending ? "Creating…" : "Create Cart"}
   163→        </Button>
   164→      </CardContent>
   165→    </Card>
   166→  );
   167→}

   320→// ─── Add Memories Dialog ────────────────────────────────────────────────────
   321→function AddMemoriesDialog({
   322→  cartId,
   323→  existingIds,
   324→}: {
   325→  cartId: string;
   326→  existingIds: Set<string>;
   327→}) {
   328→  const qc = useQueryClient();
   329→  const [open, setOpen] = React.useState(false);
   330→  const [selected, setSelected] = React.useState<Set<string>>(new Set());
   331→  const [search, setSearch] = React.useState("");
   332→
   333→  // Load all recent memories when the dialog opens.
   334→  const { data, isLoading, isError } = useQuery({
   335→    queryKey: ["memories", "all", 1, 100],
   336→    queryFn: () => api.memories.list({ page: 1, pageSize: 100 }),
   337→    enabled: open,
   338→  });
   339→
   340→  React.useEffect(() => {
   341→    if (!open) {
   342→      setSelected(new Set());
   343→      setSearch("");
   344→    }
   345→  }, [open]);
   346→
   347→  const addMut = useMutation({
   348→    mutationFn: (ids: string[]) => api.memoryCarts.patch(cartId, { addMemoryIds: ids }),
   349→    onSuccess: (_data, ids) => {
   350→      toast.success(`Added ${ids.length} memor${ids.length === 1 ? "y" : "ies"} to cart`);
   351→      qc.invalidateQueries({ queryKey: ["cart", cartId] });
   352→      qc.invalidateQueries({ queryKey: ["carts"] });
   353→      setOpen(false);
   354→    },
   355→    onError: (e: unknown) => {
   356→      const msg = e instanceof Error ? e.message : "Failed to add memories";
   357→      toast.error("Add failed", { description: msg });
   358→    },
   359→  });
   360→
   361→  const memories = (data?.items ?? []) as Memory[];
   362→  const candidates = memories.filter(
   363→    (m) => !existingIds.has(m.id) && (!search || m.queryText.toLowerCase().includes(search.toLowerCase()))
   364→  );
   365→
   366→  const toggle = (id: string) => {
   367→    setSelected((prev) => {
   368→      const next = new Set(prev);
   369→      if (next.has(id)) next.delete(id);
   370→      else next.add(id);
   371→      return next;
   372→    });
   373→  };
   374→
   375→  return (
   376→    <Dialog open={open} onOpenChange={setOpen}>
   377→      <DialogTrigger asChild>
   378→        <Button variant="outline" size="sm">
   379→          <ListPlus className="h-3.5 w-3.5" /> Add memories
   380→        </Button>
   381→      </DialogTrigger>
   382→      <DialogContent className="sm:max-w-2xl">
   383→        <DialogHeader>
   384→          <DialogTitle>Add memories to cart</DialogTitle>
   385→          <DialogDescription>
   386→            Select recent memories to add. Memories already in this cart are hidden.
   387→          </DialogDescription>
   388→        </DialogHeader>
   389→        <div className="space-y-2">
   390→          <div className="relative">
   391→            <SearchIcon className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   392→            <Input
   393→              placeholder="Filter by query text…"
   394→              value={search}
   395→              onChange={(e) => setSearch(e.target.value)}
   396→              className="pl-8 h-8 text-sm"
   397→            />
   398→          </div>
   399→          <div className="rounded-md border max-h-[50vh] overflow-y-auto thin-scroll">
   400→            {isLoading ? (
   401→              <div className="p-3 space-y-2">
   402→                {[0, 1, 2, 3].map((i) => (
   403→                  <Skeleton key={i} className="h-10 w-full" />
   404→                ))}
   405→              </div>
   406→            ) : isError ? (
   407→              <div className="p-4 text-xs text-destructive flex items-center gap-2">
   408→                <AlertCircle className="h-4 w-4" /> Failed to load memories.
   409→              </div>
   410→            ) : candidates.length === 0 ? (
   411→              <div className="p-6 text-center text-xs text-muted-foreground">
   412→                No memories available to add.
   413→              </div>
   414→            ) : (
   415→              <ul className="divide-y">
   416→                {candidates.map((m) => {
   417→                  const checked = selected.has(m.id);
   418→                  return (
   419→                    <li key={m.id}>
   420→                      <label
   421→                        className={cn(
   422→                          "flex items-start gap-2.5 p-2.5 cursor-pointer hover:bg-accent/50 transition-colors",
   423→                          checked && "bg-accent/40"
   424→                        )}
   425→                      >
   426→                        <Checkbox
   427→                          checked={checked}
   428→                          onCheckedChange={() => toggle(m.id)}
   429→                          className="mt-0.5"
   430→                          aria-label={`Select memory ${m.id}`}
   431→                        />
   432→                        <div className="min-w-0 flex-1">
   433→                          <div className="text-sm truncate">{m.queryText}</div>
   434→                          {m.chunkText && (
   435→                            <div className="text-xs text-muted-foreground line-clamp-1 mt-0.5">{m.chunkText}</div>
   436→                          )}
   437→                          <div className="text-[10px] text-muted-foreground mt-1 flex items-center gap-2">
   438→                            <span className="font-mono">score {fmtScore(m.score)}</span>
   439→                            <span>{relativeTime(m.createdAt)}</span>
   440→                          </div>
   441→                        </div>
   442→                      </label>
   443→                    </li>
   444→                  );
   445→                })}
   446→              </ul>
   447→            )}
   448→          </div>
   449→          {selected.size > 0 && (
   450→            <div className="text-xs text-muted-foreground">
   451→              {selected.size} selected
   452→            </div>
   453→          )}
   454→        </div>
   455→        <DialogFooter>
   456→          <DialogClose asChild>
   457→            <Button variant="outline" size="sm">Cancel</Button>
   458→          </DialogClose>
   459→          <Button
   460→            size="sm"
   461→            disabled={selected.size === 0 || addMut.isPending}
   462→            onClick={() => addMut.mutate(Array.from(selected))}
   463→          >
   464→            {addMut.isPending
   465→              ? "Adding…"
   466→              : `Add ${selected.size > 0 ? selected.size : ""} selected`}
   467→          </Button>
   468→        </DialogFooter>
   469→      </DialogContent>
   470→    </Dialog>
   471→  );
   472→}

   560→// ─── Memory Selection Table (cart detail) ───────────────────────────────────
   561→function MemorySelectionTable({ cartId, memories }: { cartId: string; memories: CartMemory[] }) {
   562→  const qc = useQueryClient();
   563→  const [detail, setDetail] = React.useState<CartMemory | null>(null);
   564→  const [detailOpen, setDetailOpen] = React.useState(false);
   565→
   566→  // Optimistic toggle: replace the selection with the remaining checked ids.
   567→  const toggleMut = useMutation({
   568→    mutationFn: (memoryIds: string[]) => api.memoryCarts.patch(cartId, { memoryIds }),
   569→    onMutate: async (memoryIds: string[]) => {
   570→      await qc.cancelQueries({ queryKey: ["cart", cartId] });
   571→      const prev = qc.getQueryData<{ memories: CartMemory[] }>(["cart", cartId]);
   572→      if (prev) {
   573→        const set = new Set(memoryIds);
   574→        qc.setQueryData(["cart", cartId], {
   575→          ...prev,
   576→          memories: prev.memories.filter((m) => set.has(m.id)),
   577→        });
   578→      }
   579→      return { prev };
   580→    },
   581→    onSuccess: (_data, memoryIds) => {
   582→      toast.success(`Updated selection (${memoryIds.length} memor${memoryIds.length === 1 ? "y" : "ies"})`);
   583→      qc.invalidateQueries({ queryKey: ["cart", cartId] });
   584→      qc.invalidateQueries({ queryKey: ["carts"] });
   585→    },
   586→    onError: (e: unknown, _vars, ctx) => {
   587→      if (ctx?.prev) qc.setQueryData(["cart", cartId], ctx.prev);
   588→      const msg = e instanceof Error ? e.message : "Failed to update selection";
   589→      toast.error("Update failed", { description: msg });
   590→    },
   591→  });
   592→
   593→  const handleToggle = (id: string, checked: boolean) => {
   594→    const next = memories.filter((m) => (m.id === id ? checked : true)).map((m) => m.id);
   595→    toggleMut.mutate(next);
   596→  };
   597→
   598→  const openDetail = (m: CartMemory) => {
   599→    setDetail(m);
   600→    setDetailOpen(true);
   601→  };
   602→
   603→  if (memories.length === 0) {
   604→    return (
   605→      <div className="rounded-md border border-dashed p-8 text-center">
   606→        <Inbox className="mx-auto h-8 w-8 text-muted-foreground mb-2" />
   607→        <p className="text-sm text-muted-foreground">
   608→          No memories in this cart yet.
   609→        </p>
   610→        <p className="text-xs text-muted-foreground mt-1">
   611→          Use <span className="font-medium">Add memories</span> to curate retrieval results.
   612→        </p>
   613→      </div>
   614→    );
   615→  }
   616→
   617→  return (
   618→    <>
   619→      <div className="rounded-md border max-h-[60vh] overflow-y-auto thin-scroll">
   620→        <Table>
   621→          <TableHeader className="sticky top-0 bg-background z-10">
   622→            <TableRow>
   623→              <TableHead className="w-[40px]"></TableHead>
   624→              <TableHead className="min-w-[160px]">Query</TableHead>
   625→              <TableHead className="min-w-[220px]">Chunk text</TableHead>
   626→              <TableHead className="w-[70px] text-right">Score</TableHead>
   627→              <TableHead className="w-[70px] text-right">Vector</TableHead>
   628→              <TableHead className="w-[70px] text-right">BM25</TableHead>
   629→              <TableHead className="w-[70px] text-right">Rerank</TableHead>
   630→              <TableHead className="w-[110px]">Created</TableHead>
   631→              <TableHead className="w-[40px]"></TableHead>
   632→            </TableRow>
   633→          </TableHeader>
   634→          <TableBody>
   635→            {memories.map((m) => (
   636→              <TableRow
   637→                key={m.id}
   638→                data-state={m.selected ? "selected" : undefined}
   639→                className="group"
   640→              >
   641→                <TableCell>
   642→                  <Checkbox
   643→                    checked={true}
   644→                    onCheckedChange={(v) => handleToggle(m.id, Boolean(v))}
   645→                    aria-label={`Keep memory ${m.id} in cart`}
   646→                  />
   647→                </TableCell>
   648→                <TableCell className="max-w-[200px]">
   649→                  <button
   650→                    className="text-left text-sm hover:text-primary hover:underline line-clamp-2"
   651→                    onClick={() => openDetail(m)}
   652→                  >
   653→                    {m.queryText}
   654→                  </button>
   655→                </TableCell>
   656→                <TableCell className="max-w-[260px]">
   657→                  <button
   658→                    className="text-left text-xs text-muted-foreground hover:text-primary line-clamp-2 text-left"
   659→                    onClick={() => openDetail(m)}
   660→                  >
   661→                    {m.chunkText ?? "—"}
   662→                  </button>
   663→                </TableCell>
   664→                <TableCell className="text-right font-mono text-xs">{fmtScore(m.score)}</TableCell>
   665→                <TableCell className="text-right font-mono text-xs text-muted-foreground">{fmtScore(m.vectorScore)}</TableCell>
   666→                <TableCell className="text-right font-mono text-xs text-muted-foreground">{fmtScore(m.bm25Score)}</TableCell>
   667→                <TableCell className="text-right font-mono text-xs text-muted-foreground">{fmtScore(m.rerankerScore)}</TableCell>
   668→                <TableCell className="text-[10px] text-muted-foreground whitespace-nowrap">
   669→                  {relativeTime(m.createdAt)}
   670→                </TableCell>
   671→                <TableCell>
   672→                  <Button
   673→                    variant="ghost"
   674→                    size="icon"
   675→                    className="h-6 w-6 opacity-0 group-hover:opacity-100 transition-opacity"
   676→                    onClick={() => openDetail(m)}
   677→                    aria-label="View memory detail"
   678→                  >
   679→                    <Eye className="h-3 w-3" />
   680→                  </Button>
   681→                </TableCell>
   682→              </TableRow>
   683→            ))}
   684→          </TableBody>
   685→        </Table>
   686→      </div>
   687→      <MemoryDetailSheet memory={detail} open={detailOpen} onOpenChange={setDetailOpen} />
   688→    </>
   689→  );
   690→}

   692→// ─── Cart Detail ────────────────────────────────────────────────────────────
   693→function CartDetail({ cartId }: { cartId: string }) {
   694→  const { data, isLoading, isError, refetch, isFetching } = useQuery({
   695→    queryKey: ["cart", cartId],
   696→    queryFn: () => api.memoryCarts.get(cartId),
   697→  });
   698→
   699→  if (isLoading) {
   700→    return (
   701→      <Card>
   702→        <CardHeader>
   703→          <Skeleton className="h-5 w-1/3" />
   704→          <Skeleton className="h-3 w-1/4 mt-2" />
   705→        </CardHeader>
   706→        <CardContent className="space-y-2">
   707→          {[0, 1, 2].map((i) => (
   708→            <Skeleton key={i} className="h-10 w-full" />
   709→          ))}
   710→        </CardContent>
   711→      </Card>
   712→    );
   713→  }
   714→
   715→  if (isError || !data) {
   716→    return (
   717→      <Card>
   718→        <CardContent className="pt-6">
   719→          <div className="flex items-center gap-2 text-sm text-destructive">
   720→            <AlertCircle className="h-4 w-4" /> Failed to load cart detail.
   721→          </div>
   722→          <Button variant="outline" size="sm" className="mt-3" onClick={() => refetch()}>
   723→            Retry
   724→          </Button>
   725→        </CardContent>
   726→      </Card>
   727→    );
   728→  }
   729→
   730→  const memories = (data.memories ?? []) as CartMemory[];
   731→  const existingIds = new Set(memories.map((m) => m.id));
   732→
   733→  return (
   734→    <Card>
   735→      <CardHeader>
   736→        <div className="flex items-start justify-between gap-2 min-w-0">
   737→          <div className="min-w-0">
   738→            <CardTitle className="flex items-center gap-2 text-base truncate">
   739→              <ShoppingCart className="h-4 w-4 text-primary shrink-0" />
   740→              <span className="truncate">{data.name}</span>
   741→            </CardTitle>
   742→            <CardDescription className="text-xs mt-1 flex flex-wrap items-center gap-x-2 gap-y-1">
   743→              <span>{memories.length} memor{memories.length === 1 ? "y" : "ies"}</span>
   744→              <span aria-hidden>·</span>
   745→              <span className="flex items-center gap-1">
   746→                <Clock className="h-3 w-3" />
   747→                Created {relativeTime(data.createdAt)}
   748→              </span>
   749→              <span aria-hidden>·</span>
   750→              <span>Updated {relativeTime(data.updatedAt)}</span>
   751→            </CardDescription>
   752→          </div>
   753→          <div className="flex items-center gap-1 shrink-0">
   754→            <Button
   755→              variant="ghost"
   756→              size="icon"
   757→              className="h-7 w-7"
   758→              onClick={() => refetch()}
   759→              aria-label="Refresh cart"
   760→            >
   761→              <RefreshCw className={cn("h-3.5 w-3.5", isFetching && "animate-spin")} />
   762→            </Button>
   763→            <EditCartDialog
   764→              cartId={cartId}
   765→              name={data.name}
   766→              description={data.description}
   767→            />
   768→          </div>
   769→        </div>
   770→        {data.description && (
