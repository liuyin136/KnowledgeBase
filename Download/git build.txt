# 1. 初始化 Git（如果還沒初始化）
git init

# 2. 設定遠端倉庫
git remote add origin https://github.com/liuyin136/KnowledgeBase.git

# 3. 建立並切換到 v1.3-replace 分支
git checkout -b v_template

# 4. 加入所有檔案
git add .

# 5. 提交（請修改 commit message）
git commit -m "chore: initial raw template"

# 6. 強制推送到遠端（因為是第一次上傳這個分支）
git push -u origin v_template --force
